function renderizarTabla(data) { const tbody = document.getElementById("lista-citas"); tbody.innerHTML = ""; data.forEach(c => { // Formateo de fecha: convierte YYYY-MM-DD a DD-MM-YYYY const fechaPartes = c.fecha ? c.fecha.split('-') : ['', '', '']; const fechaFormateada = fechaPartes.length === 3 ? `${fechaPartes[2]}-${fechaPartes[1]}-${fechaPartes[0]}` : c.fecha; const tr = document.createElement("tr"); tr.innerHTML = ` ${c.nombre || ''}${c.cedula || ''}${c.telefono || ''} ${c.doctor || ''} - ${c.hora || ''}${fechaFormateada || ''} ${c.confirmado ? 'Confirmado' : 'Pendiente'} ${!c.confirmado ? `` : ''} `; tbody.appendChild(tr); }); }
